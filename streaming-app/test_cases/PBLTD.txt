    /**
     * Creates mock PBL AccountBalanceEvent for successful transformation
     */
    public static List<AccountBalanceEvent> buildMockPBLAccountBalanceEvent() {
        // Create EventIdentifier
        EventIdentifier eventIdentifier = EventIdentifier.newBuilder()
                .setEventId(UUID.randomUUID().toString())
                .setEventTimestamp(Timestamp.newBuilder()
                        .setSeconds(Instant.now().getEpochSecond())
                        .setNanos(Instant.now().getNano())
                        .build())
                .build();

        // Create SourceEventIdentifier
        SourceEventIdentifier sourceEventIdentifier = SourceEventIdentifier.newBuilder()
                .setSourceEventId(generateUniqueEventId())
                .setSourceSystem(SourceSystem.VAULT_CORE)
                .setSourceEventTimestamp(Timestamp.newBuilder()
                        .setSeconds(Instant.now().getEpochSecond())
                        .setNanos(Instant.now().getNano())
                        .build())
                .build();

        // Create EventMetadata
        EventMetadata eventMetadata = EventMetadata.newBuilder()
                .setEventIdentifier(eventIdentifier)
                .setSourceEventIdentifier(sourceEventIdentifier)
                .setCorrelationId(generateUniqueEventId())
                .build();

        // Create Balance
        com.lbg.banking.accountbalance.Balance balance = com.lbg.banking.accountbalance.Balance.newBuilder()
                .setBalanceId("pbl_balance_" + UUID.randomUUID())
                .setCurrency("GBP")
                .setAmount(5000.0)
                .setType(BalanceType.LEDGER_BALANCE)
                .build();

        // Create AccountBalance
        AccountBalance accountBalance = AccountBalance.newBuilder()
                .setAccountId(PBL_ACCOUNT_ID)
                .setSortCode(PBL_SORT_CODE)
                .setAccountNumber(PBL_ACCOUNT_NUMBER)
                .setTransactionId(PBL_POSTING_BATCH_ID)
                .setTransactionDate(Timestamp.newBuilder()
                        .setSeconds(Instant.now().getEpochSecond())
                        .setNanos(Instant.now().getNano())
                        .build())
                .setBalance(balance)
                .build();

        // Build PBL AccountBalanceEvent
        AccountBalanceEvent pblEvent = AccountBalanceEvent.newBuilder()
                .setAccountBalanceEventSubject(PBL_ACCOUNT_BALANCE_EVENT_SUBJECT)
                .setEventMetadata(eventMetadata)
                .setAccountBalance(accountBalance)
                .setSourceEventTimestamp(Timestamp.newBuilder()
                        .setSeconds(Instant.now().getEpochSecond())
                        .setNanos(Instant.now().getNano())
                        .build())
                .build();

        return List.of(pblEvent);
    }

    /**
     * Creates PBL Balance Event with specific balance types for testing
     */
    public static BalanceEvent createPBLBalanceEventWithBalanceTypes(String... balanceTypes) {
        BalanceEvent event = new BalanceEvent();
        event.setEventId(generateUniqueEventId());
        event.setAccountId(PBL_ACCOUNT_ID);
        event.setPostingInstructionBatchId(PBL_POSTING_BATCH_ID);
        event.setEventTimestamp(DEFAULT_TIMESTAMP);
        event.setProductCode(PBL_PRODUCT_CODE);

        List<Balance> balances = new ArrayList<>();
        for (int i = 0; i < balanceTypes.length; i++) {
            String amount = String.valueOf((i + 1) * 100) + ".00";
            balances.add(createPBLBalance(balanceTypes[i], amount));
        }
        event.setBalances(balances);

        return event;
    }

    /**
     * Creates test data for buffer size limit testing
     */
    public static List<BalanceEvent> createMultiplePBLEventsForBuffering(int count) {
        List<BalanceEvent> events = new ArrayList<>();
        for (int i = 0; i < count; i++) {
            BalanceEvent event = validPBLBalanceEvent();
            event.setEventId(PBL_ACCOUNT_ID + "_buffering_event_" + i);
            event.setPostingInstructionBatchId(PBL_POSTING_BATCH_ID + "_" + i);
            // Update balance batch ID to match event
            event.getBalances().get(0).setPostingInstructionBatchId(event.getPostingInstructionBatchId());
            events.add(event);
        }
        return events;
    }

    /**
     * Utility method to generate unique event IDs
     */
    private static String generateUniqueEventId() {
        return PBL_ACCOUNT_ID + "_" + System.currentTimeMillis() + "_" + UUID.randomUUID().toString().substring(0, 8);
    }
